实验一　LED 灯点亮实验

实验目的

  • 了解 ESP-IDF 工程的目录结构与构建流程。
  • 掌握使用官方 GPIO 驱动控制引脚的输入输出。
  • 熟悉 FreeRTOS 的非阻塞延时函数。

实验原理

将 LED 所在引脚配置为输出模式，循环翻转其电平，每次翻转之间延时 500ms，LED 即呈现约 1Hz 的闪烁。vTaskDelay(pdMS_TO_TICKS(x)) 将毫秒换算为系统 Tick 并挂起当前任务，属于非阻塞延时，不会独占 CPU。LED 串联限流电阻的作用是限制流过 LED 的电流，防止电流过大烧毁器件。

硬件说明

器件引脚说明
外接 LEDGPIO_NUM_2（蓝）、GPIO_NUM_42（绿）、GPIO_NUM_41（红）高电平点亮，须接地形成回路
限流电阻串联在 LED 回路中限制电流，防止 LED 烧毁

参考代码

#include "driver/gpio.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#define LED_GPIO1 GPIO_NUM_2
#define LED_GPIO2 GPIO_NUM_42
#define LED_GPIO3 GPIO_NUM_41
#define LED_ON_LEVEL 1

void app_main(void)
{
    gpio_reset_pin(LED_GPIO1);
    gpio_reset_pin(LED_GPIO2);
    gpio_reset_pin(LED_GPIO3);
    gpio_set_direction(LED_GPIO1, GPIO_MODE_OUTPUT);
    gpio_set_direction(LED_GPIO2, GPIO_MODE_OUTPUT);
    gpio_set_direction(LED_GPIO3, GPIO_MODE_OUTPUT);

    while (1) {
        gpio_set_level(LED_GPIO1, LED_ON_LEVEL);
        gpio_set_level(LED_GPIO2, LED_ON_LEVEL);
        gpio_set_level(LED_GPIO3, LED_ON_LEVEL);
         vTaskDelay(pdMS_TO_TICKS(500));

        gpio_set_level(LED_GPIO1, !LED_ON_LEVEL);
        gpio_set_level(LED_GPIO2, !LED_ON_LEVEL);
        gpio_set_level(LED_GPIO3, !LED_ON_LEVEL);
        vTaskDelay(pdMS_TO_TICKS(500));
     }
}

引脚说明：本实验只做输出控制，调用 gpio_set_direction() 即可，无需 gpio_reset_pin()。GPIO 的方向、上下拉、位掩码等基础知识见 7.1 节。

实验现象

  • LED 以 500ms 为间隔交替点亮与熄灭，形成约 1Hz 的周期闪烁。
  • 修改 vTaskDelay 的参数可改变闪烁频率。
  • 本实验未启用串口输出，监视器无打印内容。

思考题

  1. 若要同时控制两个 LED（一个常亮、一个闪烁），程序应如何改写？
  1. 为什么使用 pdMS_TO_TICKS(500) 而不是直接写 500？
  1. 高电平点亮与低电平点亮的电路区别是什么？

实验二　ESP32 定时器实验

实验目的

  • 掌握 ESP-IDF 通用定时器组件的使用。
  • 理解硬件定时器中断回调与自动重载机制。
  • 对比硬件定时与软件延时在精度上的差异。

实验原理

GPTimer 可配置时钟源、计数方向与分辨率。本实验将分辨率设为 1MHz（1 tick = 1μs），闹钟计数值设为 500000（即 500ms）并开启自动重载，定时器每 500ms 触发一次中断。中断回调运行在中断上下文，不适合执行耗时 IO，因此在回调中只翻转标志位，主循环依据标志切换 GPIO 电平。

硬件说明

硬件定时器为芯片内部资源，不需要外接电路；LED 接线与实验一相同。

参考代码

#include "driver/gpio.h"
#include "driver/gptimer.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#define LED_GPIO        GPIO_NUM_2
#define LED_ON_LEVEL    1

static volatile bool led_state = false;

// GPTimer 中断回调
static bool IRAM_ATTR gptimer_callback(gptimer_handle_t timer, const gptimer_alarm_event_data_t *edata, void *user_ctx)
{
    led_state = !led_state;
    return true;
}

void app_main(void)
{
    // GPIO初始化LED
    gpio_reset_pin(LED_GPIO);
    gpio_set_direction(LED_GPIO, GPIO_MODE_OUTPUT);
    gpio_set_level(LED_GPIO, !LED_ON_LEVEL);

    // 1. 创建GPTimer实例
    gptimer_handle_t gptimer = NULL;
    gptimer_config_t timer_cfg = {
        .clk_src = GPTIMER_CLK_SRC_DEFAULT,
        .direction = GPTIMER_COUNT_UP,
        .resolution_hz = 1000 * 1000, // 1MHz，1us每tick
    };
    ESP_ERROR_CHECK(gptimer_new_timer(&timer_cfg, &gptimer));

    // 2. 注册中断回调
    gptimer_event_callbacks_t cb_cfg = {
        .on_alarm = gptimer_callback,
    };
    ESP_ERROR_CHECK(gptimer_register_event_callbacks(gptimer, &cb_cfg, NULL));

    // 3. 设置闹钟，500ms触发一次
    gptimer_alarm_config_t alarm_cfg = {
        .alarm_count = 500 * 1000,   // 500 000 us = 500ms
        .reload_count = 0,
        .flags.auto_reload_on_alarm = true, // 自动重装，循环定时
    };
    ESP_ERROR_CHECK(gptimer_set_alarm_action(gptimer, &alarm_cfg));

    // 4. 启动定时器
    ESP_ERROR_CHECK(gptimer_enable(gptimer));
    ESP_ERROR_CHECK(gptimer_start(gptimer));


    // 主循环只做IO输出，中断只改标记
    while (1)
    {
        if (led_state)
        {
            gpio_set_level(LED_GPIO, LED_ON_LEVEL);
        }
        else
        {
            gpio_set_level(LED_GPIO, !LED_ON_LEVEL);
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

实验现象

  • LED 以 500ms 为周期亮灭闪烁。
  • 相比软件延时，硬件定时精度更高且不占用 CPU 计算资源。

思考题

  1. 中断回调为什么需要使用 IRAM_ATTR 修饰？
  1. 修改 alarm_count 为其他值，闪烁周期如何变化？
  1. 若关闭 auto_reload_on_alarm，现象有何不同？

实验三　ESP32 UART 实验

实验目的

  • 了解 UART 通信的基本原理与配置项。
  • 掌握串口收发数据的编程方法。
  • 为后续通过串口打印传感器数据打下基础。

实验原理

UART 为异步串行通信，双方必须约定相同的波特率与帧格式。板载 USB 转串口芯片连接 UART0，printf 的输出重定向到该端口。业务串口使用 UART1 或 UART2。接收方式分为轮询与中断两种：轮询简单直观，中断实时性更好。

硬件说明

器件说明
杜邦线将 GPIO16 与 GPIO17 短接，实现自收发测试

参考代码

#include <stdio.h>
#include <string.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/uart.h"
#include "driver/gpio.h"

// 更改UART_RECV_MODE的值来开启不同的功能（值 0 ， 1）  需要短接16，17引脚
//  0 = 查询(轮询)接收方式 代码循环主动去问串口硬件有没有收到数据
//  1 = 中断+队列接收方式  串口收到 1 字节数据，立刻触发 UART 硬件中断；CPU 立刻停下当前工作去接收数据

#define UART_RECV_MODE 1

#define UART_NUM UART_NUM_2
#define UART_TX_PIN (GPIO_NUM_16)
#define UART_RX_PIN (GPIO_NUM_17)
#define UART_BUF_SIZE (1024)
#define UART_BAUD_RATE (115200)

static QueueHandle_t uart_rx_queue;
// 初始化UART(串口)
static void uart_init(void)
{
    uart_config_t uart_config = {
        .baud_rate = UART_BAUD_RATE,
        .data_bits = UART_DATA_8_BITS,
        .parity = UART_PARITY_DISABLE,
        .stop_bits = UART_STOP_BITS_1,
        .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
        .source_clk = UART_SCLK_DEFAULT,
    };
    ESP_ERROR_CHECK(uart_param_config(UART_NUM, &uart_config));
    ESP_ERROR_CHECK(uart_set_pin(UART_NUM, UART_TX_PIN, UART_RX_PIN,
                                 UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE));
    gpio_set_pull_mode(UART_RX_PIN, GPIO_PULLUP_ONLY);

#if (UART_RECV_MODE == 1)
    ESP_ERROR_CHECK(uart_driver_install(UART_NUM, UART_BUF_SIZE, UART_BUF_SIZE,
                                        10, &uart_rx_queue, 0));
#else
    ESP_ERROR_CHECK(uart_driver_install(UART_NUM, UART_BUF_SIZE, UART_BUF_SIZE,
                                        0, NULL, 0));
#endif
}

// 轮询模式
static void uart_poll_rx_task(void *arg)
{
    uint8_t data[UART_BUF_SIZE];
    while (1)
    {
        int len = uart_read_bytes(UART_NUM, data, sizeof(data), 0);
        if (len > 0)
        {
            printf("[查询方式]收到数据,长度=%d\r\n", len);
        }
        else
        {
            printf("正在轮询查询串口\r\n");
        }
        vTaskDelay(pdMS_TO_TICKS(200));
    }
}

// 中断
static void uart_rx_task(void *arg)
{
    uint8_t data[UART_BUF_SIZE];
    uart_event_t event;
    while (1)
    {
        // 永久阻塞
        if (xQueueReceive(uart_rx_queue, &event, portMAX_DELAY))
        {
            switch (event.type)
            {
            case UART_DATA:
            {
                int len = uart_read_bytes(UART_NUM, data, sizeof(data),
                                          pdMS_TO_TICKS(20));
                if (len > 0)
                {
                    printf("[中断方式]收到中断！数据长度 = %d\r\n", len);
                }
                break;
            }
            case UART_FIFO_OVF:
                uart_flush_input(UART_NUM);
                xQueueReset(uart_rx_queue);
                break;
            case UART_BUFFER_FULL:
                uart_flush_input(UART_NUM);
                xQueueReset(uart_rx_queue);
                break;
            default:
                break;
            }
        }
    }
}

void app_main(void)
{
    uart_init();

#if (UART_RECV_MODE == 1)
    xTaskCreate(uart_rx_task, "uart_rx_task", 4096, NULL, 10, NULL);
#else
    xTaskCreate(uart_poll_rx_task, "uart_poll_rx_task", 4096, NULL, 10, NULL);
#endif

    // 每隔1秒发送一包测试数据
    while (1)
    {
        uart_write_bytes(UART_NUM, "test\r\n", strlen("test\r\n"));
        printf("UART2 已经发送 test\r\n");
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

  • 串口监视器每秒打印一次发送与接收的内容。
  • 短接 GPIO16 与 GPIO17 后，发送的数据被自身接收，可验证串口配置正确。

  1. 轮询接收与中断接收各适用于什么场景？
  1. 串口通信为什么必须共地？

实验四　光敏 ADC 采集实验

实验目的

  • 掌握 ADC 的配置方法与采样流程。
  • 理解分辨率、量程与电压换算的关系。
  • 使用光敏传感器实现光照强度检测与阈值控制。

实验原理

ADC 为 12 位逐次逼近型，原始值范围 0~4095。通过衰减档位扩展量程，ADC_ATTEN_DB_12 对应满量程约 2450mV。电压换算公式为 电压 = 原始值 × 满量程 ÷ 4095。光照越强，光敏电阻阻值越小，分压点电压随之变化。

硬件说明

器件引脚 / 通道说明
光敏传感器信号ADC1_CH0（约 GPIO1）输出电压随光照变化
指示 LEDGPIO_NUM_2高电平点亮

参考代码

工程配置：该实验的 main/CMakeLists.txt 需要声明 REQUIRES esp_adc。

#include <stdio.h>
#include "driver/gpio.h"
#include "esp_adc/adc_oneshot.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

/**
 * @brief   光敏传感器（光照强度）ADC 采集实验
 * @note    光敏传感器信号接至 ADC1 通道0。
 *          在 ESP32-S3 上，ADC1_CH0 对应 GPIO1，请按实际接线确认。
 */
#define LIGHT_ADC_CHANNEL ADC_CHANNEL_0
#define LED_GPIO GPIO_NUM_2
#define GPIO_LEVEL 1

void app_main(void)
{
    // LED初始化
    gpio_reset_pin(LED_GPIO);
    gpio_set_direction(LED_GPIO, GPIO_MODE_OUTPUT);

    /* ========== ADC（光敏）初始化 ========== */
    adc_oneshot_unit_handle_t adc_handle = NULL;

    adc_oneshot_unit_init_cfg_t init_cfg = {
        .unit_id = ADC_UNIT_1,
    };
    ESP_ERROR_CHECK(adc_oneshot_new_unit(&init_cfg, &adc_handle));

    adc_oneshot_chan_cfg_t chan_cfg = {
        .atten = ADC_ATTEN_DB_12,   /* 量程 0~3.3V，最灵敏 */
        .bitwidth = ADC_BITWIDTH_12, /* 12 位分辨率，0~4095 */
    };
    ESP_ERROR_CHECK(adc_oneshot_config_channel(adc_handle, LIGHT_ADC_CHANNEL, &chan_cfg));

    printf("光敏传感器 ADC 采集实验已启动\n");

    while (1)
    {
        int light_raw = 0;
        uint32_t light_mv = 0;

        /* 读取一次光敏 ADC 原始值 */
        if (adc_oneshot_read(adc_handle, LIGHT_ADC_CHANNEL, &light_raw) == ESP_OK)
        {
            /* 由原始值换算电压：raw * 3300 / 4095（mV） */
            light_mv = (uint32_t)light_raw * 3300 / 4095;
            printf("光照强度 - ADC_raw: %d, 电压: %lu mV\n", light_raw, (unsigned long)light_mv);
        }
        else
        {
            printf("读取光敏 ADC 失败\n");
        }
        if(light_raw <= 1000)
        {
            gpio_set_level(LED_GPIO,GPIO_LEVEL);     
            printf("LED:TURN ON\n");
        }
        else
        {
             gpio_set_level(LED_GPIO,!GPIO_LEVEL);
            printf("LED:TURN OFF\n");
        }

        /* 每 1 秒采集一次 */
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

实验现象

  • 串口每秒打印一次原始值与换算电压。
  • 遮挡或照射光敏传感器时数值随之变化。
  • 光照强度低于阈值 1000 时 LED 点亮，高于阈值时熄灭。

思考题

  1. ADC 12 位分辨率的取值范围为什么是 0~4095？
  1. ADC_ATTEN_DB_12 的含义是什么？改变档位对量程有何影响？
  1. 为什么需要做 raw * 2450 / 4095 的换算？
  1. 若光敏输出接反（光照越强电压越低），判断逻辑应如何调整？

实验五　L298N 电机驱动实验

实验目的

  • 掌握 LEDC 硬件 PWM 的配置方法。
  • 理解 L298N 的转向控制与调速原理。
  • 实现电机由 0% 缓慢升至 100%、再降至 0% 的循环调速。

实验原理

L298N 为双路 H 桥驱动芯片：IN1、IN2 控制 A 通道电机转向，IN3、IN4 控制 B 通道转向，改变四路电平组合可实现前进、后退、转向与停止；ENA、ENB 接收 PWM 信号调节电机平均电压，占空比越大转速越高。

PWM 为等幅、占空比可变的方波。设占空比为 a，周期为 T，电枢两端最大电压为 U，则平均电压 Uo = a × U。在转矩不变的前提下，电机转速与供电电压成正比，因此改变占空比即可调速。三种改变占空比的方法中，定频调宽法保持频率不变、只改变高低电平时间，避免了频率接近机械固有频率引起的振荡，工程中最常采用。

硬件说明

器件引脚说明
L298N IN1GPIO_NUM_47A 通道控制输入 1
L298N IN2GPIO_NUM_21A 通道控制输入 2
L298N IN3GPIO_NUM_20B 通道控制输入 3
L298N IN4GPIO_NUM_19B 通道控制输入 4
电机由 OUT1~OUT4 驱动需独立 12V 电源供电

引脚说明：IN1~IN4 按本开发板接线固定为 GPIO47 / GPIO21 / GPIO20 / GPIO19；ENA、ENB 用于输出 PWM，示例中使用 GPIO14 与 GPIO38（避开 GPIO0 / GPIO45 / GPIO46 等启动配置引脚）。若实际接线不同，请修改宏定义。

main/main.c

#include <stdio.h>
#include "motor.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

/**
 * @brief   L298N 电机驱动控制实验
 * @note    先演示 PWM 调速（前进 0%→100%→0%），
 *          再依次演示：前进 → 后退 → 左转 → 右转 → 停止
 *          每个动作持续 3 秒，动作之间停止 1 秒，便于观察。
 */
void app_main(void)
{
    motor_init();
    printf("========== L298N 电机驱动控制实验开始 ==========\n");

    while (1)
    {
        /* ---- 前进：20秒 0% → 100% ---- */
        motor_run(MOTOR_FORWARD);
        printf("[调速] 开始20s升速 0%% →100%%\n");
        for (int speed = 0; speed <= 100; speed += 1)
        {
            motor_set_speed(speed);
            printf("[调速] 前进 速度 %d%%\n", speed);
            vTaskDelay(pdMS_TO_TICKS(200));
        }
        printf("[调速] 到达100%%\n");

        /* ---- 前进：20秒 100% → 0% ---- */
        printf("[调速] 开始20s降速 100%% →0%%\n");
        for (int speed = 100; speed >= 0; speed -= 1)
        {
            motor_set_speed(speed);
            printf("[调速] 前进 速度 %d%%\n", speed);
            vTaskDelay(pdMS_TO_TICKS(200));
        }
        printf("[调速] 降到0%%\n");

        motor_run(MOTOR_STOP);
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

/* ========== PWM 调速参数 ========== */
#define MOTOR_PWM_MODE  LEDC_LOW_SPEED_MODE /* 低速模式 */
#define MOTOR_PWM_TIMER LEDC_TIMER_0        /* 定时器 */
#define MOTOR_PWM_CH_A  LEDC_CHANNEL_0      /* ENA 通道 */
#define MOTOR_PWM_CH_B  LEDC_CHANNEL_1      /* ENB 通道 */
#define MOTOR_PWM_RES   LEDC_TIMER_8_BIT    /* 8 位分辨率（0~255） */
#define MOTOR_PWM_FREQ  1000                /* 1kHz */

/**
 * @brief       初始化电机控制引脚为推挽输出，并初始化 ENA/ENB 的 PWM
 * @retval      无
 */
void motor_init(void)
{
    gpio_config_t io_conf = {
        .intr_type = GPIO_INTR_DISABLE,
        .mode = GPIO_MODE_OUTPUT,
        .pin_bit_mask = (1ULL << MOTOR_IN1) |
                        (1ULL << MOTOR_IN2) |
                        (1ULL << MOTOR_IN3) |
                        (1ULL << MOTOR_IN4),
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .pull_up_en = GPIO_PULLUP_DISABLE,
    };
    gpio_config(&io_conf);

    /* 初始全部拉低 → 电机停止 */
    gpio_set_level(MOTOR_IN1, 0);
    gpio_set_level(MOTOR_IN2, 0);
    gpio_set_level(MOTOR_IN3, 0);
    gpio_set_level(MOTOR_IN4, 0);

    /* 初始化 PWM 定时器（ENA/ENB 调速共用） */
    ledc_timer_config_t timer_conf = {
        .speed_mode = MOTOR_PWM_MODE,
        .timer_num = MOTOR_PWM_TIMER,
        .duty_resolution = MOTOR_PWM_RES,
        .freq_hz = MOTOR_PWM_FREQ,
        .clk_cfg = LEDC_AUTO_CLK,
    };
    ledc_timer_config(&timer_conf);

    /* 通道 A → ENA */
    ledc_channel_config_t ch_a = {
        .speed_mode = MOTOR_PWM_MODE,
        .channel = MOTOR_PWM_CH_A,
        .timer_sel = MOTOR_PWM_TIMER,
        .intr_type = LEDC_INTR_DISABLE,
        .gpio_num = MOTOR_ENA,
        .duty = 0,
        .hpoint = 0,
    };
    ledc_channel_config(&ch_a);

    /* 通道 B → ENB */
    ledc_channel_config_t ch_b = {
        .speed_mode = MOTOR_PWM_MODE,
        .channel = MOTOR_PWM_CH_B,
        .timer_sel = MOTOR_PWM_TIMER,
        .intr_type = LEDC_INTR_DISABLE,
        .gpio_num = MOTOR_ENB,
        .duty = 0,
        .hpoint = 0,
    };
    ledc_channel_config(&ch_b);

    printf("电机驱动 GPIO + PWM 初始化完成\n");
}

/**
 * @brief       通过设置 L298N 的 IN1~IN4 电平控制电机动作方向
 * @param       dir: 动作方向 MOTOR_STOP/FORWARD/BACK/LEFT/RIGHT
 * @retval      无
 *
 * L298N 连接方式（以 小车电机通道 A、B 为例）：
 *   IN1/IN2 控制通道 A，IN3/IN4 控制通道 B。
 *   两通道同时正转 → 前进；同时反转 → 后退；差分 → 左转/右转。
 */
void motor_run(motor_dir_t dir)
{
    switch (dir)
    {
    case MOTOR_FORWARD: /* 前进 */
        gpio_set_level(MOTOR_IN1, 1);
        gpio_set_level(MOTOR_IN2, 0);
        gpio_set_level(MOTOR_IN3, 1);
        gpio_set_level(MOTOR_IN4, 0);
        printf("[电机] 前进\n");
        break;

    case MOTOR_BACK: /* 后退 */
        gpio_set_level(MOTOR_IN1, 0);
        gpio_set_level(MOTOR_IN2, 1);
        gpio_set_level(MOTOR_IN3, 0);
        gpio_set_level(MOTOR_IN4, 1);
        printf("[电机] 后退\n");
        break;

    case MOTOR_LEFT: /* 左转 */
        gpio_set_level(MOTOR_IN1, 1);
        gpio_set_level(MOTOR_IN2, 0);
        gpio_set_level(MOTOR_IN3, 0);
        gpio_set_level(MOTOR_IN4, 1);
        printf("[电机] 左转\n");
        break;

    case MOTOR_RIGHT: /* 右转 */
        gpio_set_level(MOTOR_IN1, 0);
        gpio_set_level(MOTOR_IN2, 1);
        gpio_set_level(MOTOR_IN3, 1);
        gpio_set_level(MOTOR_IN4, 0);
        printf("[电机] 右转\n");
        break;

    case MOTOR_STOP: /* 停止 */
    default:
        gpio_set_level(MOTOR_IN1, 0);
        gpio_set_level(MOTOR_IN2, 0);
        gpio_set_level(MOTOR_IN3, 0);
        gpio_set_level(MOTOR_IN4, 0);
        printf("[电机] 停止\n");
        break;
    }
}

/**
 * @brief       通过 PWM 设置电机转速（同时作用于 ENA、ENB）
 * @param       percent: 速度百分比 0~100，0 为停止，100 为全速
 * @retval      无
 */
void motor_set_speed(int percent)
{
    if (percent < 0)
        percent = 0;
    if (percent > 100)
        percent = 100;

    /* 8 位分辨率：占空比 0~255 */
    uint32_t duty = (uint32_t)(percent * 255 / 100);

    ledc_set_duty(MOTOR_PWM_MODE, MOTOR_PWM_CH_A, duty);
    ledc_update_duty(MOTOR_PWM_MODE, MOTOR_PWM_CH_A);

    ledc_set_duty(MOTOR_PWM_MODE, MOTOR_PWM_CH_B, duty);
    ledc_update_duty(MOTOR_PWM_MODE, MOTOR_PWM_CH_B);
}

/**
 * @brief UDP服务器任务
 */
static void udp_server_task(void *pvParameters)
{
    int server_fd;
    struct sockaddr_in server_addr, client_addr;
    socklen_t client_len = sizeof(client_addr);
    char rx_buffer[128];

    // 1. 创建socket（UDP）
    server_fd = socket(AF_INET, SOCK_DGRAM, 0);
    if (server_fd < 0)
    {
        ESP_LOGE(TAG, "socket 创建失败");
        vTaskDelete(NULL);
        return;
    }

    // 2. 绑定端口
    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = htonl(INADDR_ANY);
    server_addr.sin_port = htons(UDP_SERVER_PORT);

    if (bind(server_fd, (struct sockaddr *)&server_addr, sizeof(server_addr)) < 0)
    {
        ESP_LOGE(TAG, "bind 端口 %d 失败", UDP_SERVER_PORT);
        close(server_fd);
        vTaskDelete(NULL);
        return;
    }

    ESP_LOGI(TAG, "UDP服务器已启动，端口：%d", UDP_SERVER_PORT);

    while (1)
    {
        // 3. 阻塞接收数据，并记录发送方地址
        int len = recvfrom(server_fd, rx_buffer, sizeof(rx_buffer) - 1, 0,
                           (struct sockaddr *)&client_addr, &client_len);
        if (len > 0)
        {
            rx_buffer[len] = '\0';
            ESP_LOGI(TAG, "收到客户端消息(%s:%d): %s",
                     inet_ntoa(client_addr.sin_addr), ntohs(client_addr.sin_port), rx_buffer);
            // 回发数据给发送方
            sendto(server_fd, "OK\r\n", 4, 0,
                   (struct sockaddr *)&client_addr, client_len);
        }
        else if (len < 0)
        {
            ESP_LOGE(TAG, "接收数据失败");
        }
    }
    close(server_fd);
    vTaskDelete(NULL);
}

/**
 * @brief TCP服务器任务
 */
static void tcp_server_task(void *pvParameters)
{
    int server_fd, client_fd;
    struct sockaddr_in server_addr, client_addr;
    socklen_t client_len = sizeof(client_addr);
    char rx_buffer[128];

    // 1. 创建socket（TCP）
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd < 0)
    {
        ESP_LOGE(TAG, "socket 创建失败");
        vTaskDelete(NULL);
        return;
    }

    // 2. 绑定端口
    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = htonl(INADDR_ANY);
    server_addr.sin_port = htons(TCP_SERVER_PORT);

    if (bind(server_fd, (struct sockaddr *)&server_addr, sizeof(server_addr)) < 0)
    {
        ESP_LOGE(TAG, "bind 端口 %d 失败", TCP_SERVER_PORT);
        close(server_fd);
        vTaskDelete(NULL);
        return;
    }

    // 3. 开启监听，等待客户端连接
    listen(server_fd, 1);
    ESP_LOGI(TAG, "TCP服务器已启动，端口：%d", TCP_SERVER_PORT);

    while (1)
    {
        // 4. 阻塞等待客户端接入
        client_fd = accept(server_fd, (struct sockaddr *)&client_addr, &client_len);
        if (client_fd >= 0)
        {
            ESP_LOGI(TAG, "客户端已接入: %s:%d",
                     inet_ntoa(client_addr.sin_addr), ntohs(client_addr.sin_port));
            while (1)
            {
                // 接收电脑发来的数据
                int len = recv(client_fd, rx_buffer, sizeof(rx_buffer) - 1, 0);
                if (len > 0)
                {
                    rx_buffer[len] = '\0';
                    ESP_LOGI(TAG, "收到客户端消息: %s", rx_buffer);
                    // 回发数据给电脑
                    send(client_fd, "OK\r\n", 4, 0);
                }
                else if (len == 0 || len < 0)
                {
                    ESP_LOGI(TAG, "客户端断开连接");
                    close(client_fd);
                    break;
                }
            }
        }
    }
    close(server_fd);
    vTaskDelete(NULL);
}

实验七　ESP32 WiFi STA 模式实验

实验目的

  • 掌握 STA 模式下连接路由器的方法。
  • 理解 DHCP 自动获取 IP 的过程。
  • 实现设备接入局域网并与上位机通信。

实验原理

STA 模式下设备作为站点接入已有无线网络，通过 DHCP 自动获取 IP 地址。连接过程是异步的：esp_wifi_connect() 发起连接后立即返回，实际结果通过 WIFI_EVENT_STA_CONNECTED 与 IP_EVENT_STA_GOT_IP 事件通知。连接失败时需在事件回调中重新发起连接。

参考代码

#include <string.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/event_groups.h"
#include "esp_system.h"
#include "esp_wifi.h"
#include "esp_event.h"
#include "esp_log.h"
#include "esp_netif.h"
#include "nvs_flash.h"

static const char *TAG = "ESP32_STA";

/* WiFi 配置 —— 改成你自己的 */
#define EXAMPLE_WIFI_SSID     "your_ssid"
#define EXAMPLE_WIFI_PASS     "your_password"

/* 事件组位定义 */
#define WIFI_CONNECTED_BIT    BIT0
#define WIFI_FAIL_BIT         BIT1

static EventGroupHandle_t s_wifi_event_group;

/* WiFi 事件处理 */
static void wifi_event_handler(void* arg, esp_event_base_t event_base,
                               int32_t event_id, void* event_data)
{
    if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_START) {
        esp_wifi_connect();
        ESP_LOGI(TAG, "STA start, connecting...");

    } else if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_DISCONNECTED) {
        ESP_LOGW(TAG, "Disconnected, retrying...");
        esp_wifi_connect();

    } else if (event_base == IP_EVENT && event_id == IP_EVENT_STA_GOT_IP) {
        ip_event_got_ip_t* event = (ip_event_got_ip_t*) event_data;
        ESP_LOGI(TAG, "Got IP: " IPSTR, IP2STR(&event->ip_info.ip));
        xEventGroupSetBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
    }
}

void wifi_init_sta(void)
{
    s_wifi_event_group = xEventGroupCreate();

    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());
    esp_netif_create_default_wifi_sta();

    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));

    ESP_ERROR_CHECK(esp_event_handler_instance_register(
        WIFI_EVENT,
        ESP_EVENT_ANY_ID,
        &wifi_event_handler,
        NULL,
        NULL));

    ESP_ERROR_CHECK(esp_event_handler_instance_register(
        IP_EVENT,
        IP_EVENT_STA_GOT_IP,
        &wifi_event_handler,
        NULL,
        NULL));

    wifi_config_t wifi_config = {
        .sta = {
            .ssid = EXAMPLE_WIFI_SSID,
            .password = EXAMPLE_WIFI_PASS,
            .threshold.authmode = WIFI_AUTH_WPA2_PSK,
            .sae_pwe_h2e = WPA3_SAE_PWE_BOTH,
        },
    };
    /* 清除 ssid/password 的尾部垃圾（重要） */
    memset(wifi_config.sta.ssid + strlen(EXAMPLE_WIFI_SSID), 0,
           sizeof(wifi_config.sta.ssid) - strlen(EXAMPLE_WIFI_SSID));
    memset(wifi_config.sta.password + strlen(EXAMPLE_WIFI_PASS), 0,
           sizeof(wifi_config.sta.password) - strlen(EXAMPLE_WIFI_PASS));

    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
    ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_STA, &wifi_config));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_LOGI(TAG, "wifi_init_sta finished, connecting to %s", EXAMPLE_WIFI_SSID);

    /* 等待连接结果（最多 15 秒） */
    EventBits_t bits = xEventGroupWaitBits(
        s_wifi_event_group,
        WIFI_CONNECTED_BIT | WIFI_FAIL_BIT,
        pdFALSE,
        pdFALSE,
        pdMS_TO_TICKS(15000));

    if (bits & WIFI_CONNECTED_BIT) {
        ESP_LOGI(TAG, "Connected to AP successfully!");
    } else {
        ESP_LOGE(TAG, "Failed to connect to AP");
    }
}

实验九　DHT11 温湿度采集实验

实验目的

  • 掌握单总线协议的时序判定方法。
  • 理解微秒级延时与任务延时的区别。
  • 实现温湿度数据的周期采集与校验。

实验原理

DHT11 使用单根数据线通信，通过高低电平持续时间区分数据位。主机先拉低总线至少 18ms 发起始信号，随后释放并等待从机响应；从机依次返回 40 位数据（湿度整数、湿度小数、温度整数、温度小数、校验和），前四字节之和的低 8 位应等于校验和。该协议对时序敏感，微秒延时必须使用 esp_rom_delay_us()。

#include "dht11.h"

/**
 * @brief       复位DHT11
 * @param       无
 * @retval      无
 */
void dht11_reset(void)
{
    DHT11_DQ_OUT(0);         /* 拉低DQ */
    esp_rom_delay_us(25000); /* 拉低至少18ms */
    DHT11_DQ_OUT(1);         /* DQ=1 */
    esp_rom_delay_us(30);    /* 主机拉高10~35us */
}

/**
 * @brief       等待DHT11的回应
 * @param       无
 * @retval      0,DHT11正常; 1,DHT11异常/不存在
 */
uint8_t dht11_check(void)
{
    uint8_t retry = 0;
    uint8_t rval = 0;

    while (DHT11_DQ_IN && retry < 100) /* DHT11会拉低40~80us */
    {
        retry++;
        esp_rom_delay_us(1);
    }

    if (retry >= 100)
    {
        rval = 1;
    }
    else
    {
        retry = 0;

        while (!DHT11_DQ_IN && retry < 100) /* DHT11拉低后会再次拉高87us */
        {
            retry++;
            esp_rom_delay_us(1);
        }

        if (retry >= 100)
        {
            rval = 1;
        }
    }

    return rval;
}

/**
 * @brief       从DHT11读取一个位
 * @param       无
 * @retval      读取到的位值: 0 / 1
 */
uint8_t dht11_read_bit(void)
{
    uint8_t retry = 0;

    while (DHT11_DQ_IN && retry < 100) /* 等待变为低电平 */
    {
        retry++;
        esp_rom_delay_us(1);
    }

    retry = 0;

    while (!DHT11_DQ_IN && retry < 100) /* 等待变高电平 */
    {
        retry++;
        esp_rom_delay_us(1);
    }

    esp_rom_delay_us(40); /* 等待40us */

    if (DHT11_DQ_IN) /* 根据引脚状态返回 bit */
    {
        return 1;
    }
    else
    {
        return 0;
    }
}

/**
 * @brief       从DHT11读取一个字节
 * @param       无
 * @retval      data：读到的数据
 */
static uint8_t dht11_read_byte(void)
{
    uint8_t i, data = 0;

    for (i = 0; i < 8; i++) /* 循环读取8位数据 */
    {
        data <<= 1;               /* 高位数据先输出, 先左移一位 */
        data |= dht11_read_bit(); /* 读取1bit数据 */
    }

    return data;
}

/**
 * @brief       从DHT11读取一次数据(温度和湿度的整数部分)
 * @param       temp: 温度值(范围:-20~60°)(放大10倍)
 * @param       humi: 湿度值(范围:5%~95%)(放大10倍)
 * @retval      0, 正常; 1, 失败
 */
uint8_t dht11_read_data(short *temp, short *humi)
{
    uint8_t buf[5];
    uint8_t i;
    short raw_temp = 0;
    short raw_humi = 0;

    dht11_reset();

    if (dht11_check() == 0)
    {
        for (i = 0; i < 5; i++) /* 读取40位数据 buf[0]湿度高8位 buf[1]湿度低8位 buf[2]温度高8位 buf[3]温度低8位 */
        {
            buf[i] = dht11_read_byte();
        }

        if ((buf[0] + buf[1] + buf[2] + buf[3]) == buf[4]) /* 数据校验正确 */
        {
            raw_humi = buf[0] * 10 + buf[1]; /* 获取湿度数据 */

            if (buf[3] & 0x80) /* 温度为负值 */
            {
                raw_temp = buf[2] * 10 + (buf[3] & 0x7F);
                raw_temp = -raw_temp;
            }
            else
            {
                raw_temp = buf[2] * 10 + buf[3]; /* 温度数据 */
            }

            *humi = raw_humi;
            *temp = raw_temp;
            return 0; /* 读取成功 */
        }

        return 1; /* 校验和错误 */
    }

    return 1; /* DHT11未响应 */
}

/**
 * @brief       初始化DHT11
 * @param       无
 * @retval      0, 正常; 1, 不存在/不正常
 */
uint8_t dht11_init(void)
{
    gpio_config_t gpio_init_struct;

    gpio_init_struct.intr_type = GPIO_INTR_DISABLE;            /* 失能引脚中断 */
    gpio_init_struct.mode = GPIO_MODE_INPUT_OUTPUT_OD;         /* 开漏模式的输入和输出 */
    gpio_init_struct.pull_up_en = GPIO_PULLUP_ENABLE;          /* 使能上拉 */
    gpio_init_struct.pull_down_en = GPIO_PULLDOWN_DISABLE;     /* 失能下拉 */
    gpio_init_struct.pin_bit_mask = 1ull << DHT11_DQ_GPIO_PIN; /* 设置的引脚的位掩码 */
    gpio_config(&gpio_init_struct);                            /* 配置DHT11引脚 */

    dht11_reset();
    return dht11_check();
}

}

/* /stream → MJPEG 实时流 */
static esp_err_t stream_handler(httpd_req_t *req)
{
    esp_err_t res = httpd_resp_set_type(req, STREAM_CONTENT);
    if (res != ESP_OK)
        return res;

    printf("MJPEG 客户端已连接\n");

    while (true)
    {
        camera_fb_t *fb = esp_camera_fb_get();
        if (!fb)
        {
            printf("stream: 取帧失败\n");
            httpd_resp_sendstr_chunk(req, "--" STREAM_BOUNDARY "--\r\n");
            break;
        }

        char hdr[160];
        int hdr_len = snprintf(hdr, sizeof(hdr),
                               "--" STREAM_BOUNDARY "\r\n"
                               "Content-Type: image/jpeg\r\n"
                               "Content-Length: %u\r\n\r\n",
                               fb->len);

        if (httpd_resp_send_chunk(req, hdr, hdr_len) != ESP_OK || httpd_resp_send_chunk(req, (const char *)fb->buf, fb->len) != ESP_OK)
        {
            esp_camera_fb_return(fb);
            break;
        }
        esp_camera_fb_return(fb);
    }

    httpd_resp_sendstr_chunk(req, NULL);
    printf("MJPEG 客户端断开\n");
    return ESP_OK;
}

/* ========== 启动 Web 服务器 ========== */
void start_webserver(void)
{
    httpd_handle_t server = NULL;
    httpd_config_t config = HTTPD_DEFAULT_CONFIG();
    config.server_port = 80;
    config.lru_purge_enable = true;
    config.max_uri_handlers = 8;

    if (httpd_start(&server, &config) != ESP_OK)
    {
        printf("HTTP 服务器启动失败\n");
        return;
    }

    httpd_uri_t root = {.uri = "/", .method = HTTP_GET, .handler = root_handler};
    httpd_uri_t str = {.uri = "/stream", .method = HTTP_GET, .handler = stream_handler};
    httpd_register_uri_handler(server, &root);
    httpd_register_uri_handler(server, &str);

    printf("HTTP 服务器已启动，请在浏览器访问 http://<IP 地址>/\n");
}

void app_main(void)
{
    /* 初始化摄像头 */
    camera_init();

    /* 联网并启动视频服务器 */
    wifi_init();
    start_webserver();

    while (1)
    {
        vTaskDelay(pdMS_TO_TICKS(10000));
    }
}
