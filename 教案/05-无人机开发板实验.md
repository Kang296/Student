## 十、无人机开发板实验

> 平台资源：ICM20602 六轴惯性传感器（I2C：SDA 41 / SCL 42）、PMW3901 光流传感器（SPI：SCLK 7 / MOSI 15 / MISO 16 / NCS 6）、SPL06-001 气压计（SPI：SCK 7 / SDA 15 / SDO 16 / CSB 8）、VL53L0X 激光测距（I2C：SCL 42 / SDA 41）、IMX219 摄像头、板载 WiFi 与 BLE。

### 实验一　ESP32 WiFi AP 模式实验

#### 实验目的

- 掌握 AP 模式热点的创建与参数配置。
- 掌握基于 UDP / TCP 的数据收发流程。
- 实现无人机平台与上位机的无线数据交互通道。

#### 实验原理

AP 模式下设备自身作为无线接入点。无人机的飞行数据通过 UDP 周期上报，控制指令通过 TCP 可靠下发，两条链路分别满足"高频低延迟"与"可靠不丢包"两类需求。事件回调用于监控设备接入状态。

```c
static void wifi_event_handler(void *arg, esp_event_base_t event_base,
                               int32_t event_id, void *event_data)
{
    if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_AP_STACONNECTED)
    {
        wifi_event_ap_staconnected_t *event =
            (wifi_event_ap_staconnected_t *)event_data;

                 event->mac[3], event->mac[4], event->mac[5],

                 event->mac[3], event->mac[4], event->mac[5],
                 event->aid);
    }

/**
 * @brief UDP服务器任务
 */
static void udp_server_task(void *pvParameters)
{
    int server_fd;
    struct sockaddr_in server_addr, client_addr;
    socklen_t client_len = sizeof(client_addr);
    char rx_buffer[128];

    // 1. 创建socket（UDP）
    server_fd = socket(AF_INET, SOCK_DGRAM, 0);
    if (server_fd < 0)
    {
        ESP_LOGE(TAG, "socket 创建失败");
        vTaskDelete(NULL);
        return;
    }

    // 2. 绑定端口
    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = htonl(INADDR_ANY);
    server_addr.sin_port = htons(UDP_SERVER_PORT);

    if (bind(server_fd, (struct sockaddr *)&server_addr, sizeof(server_addr)) < 0)
    {
        ESP_LOGE(TAG, "bind 端口 %d 失败", UDP_SERVER_PORT);
        close(server_fd);
        vTaskDelete(NULL);
        return;
    }

    ESP_LOGI(TAG, "UDP服务器已启动，端口：%d", UDP_SERVER_PORT);

    while (1)
    {
        // 3. 阻塞接收数据，并记录发送方地址
        int len = recvfrom(server_fd, rx_buffer, sizeof(rx_buffer) - 1, 0,
                           (struct sockaddr *)&client_addr, &client_len);
        if (len > 0)
        {
            rx_buffer[len] = '\0';
            ESP_LOGI(TAG, "收到客户端消息(%s:%d): %s",
                     inet_ntoa(client_addr.sin_addr), ntohs(client_addr.sin_port), rx_buffer);
            // 回发数据给发送方
            sendto(server_fd, "OK\r\n", 4, 0,
                   (struct sockaddr *)&client_addr, client_len);
        }
        else if (len < 0)
        {
            ESP_LOGE(TAG, "接收数据失败");
        }
    }
    close(server_fd);
    vTaskDelete(NULL);
}

/**
 * @brief TCP服务器任务
 */
static void tcp_server_task(void *pvParameters)
{
    int server_fd, client_fd;
    struct sockaddr_in server_addr, client_addr;
    socklen_t client_len = sizeof(client_addr);
    char rx_buffer[128];

    // 1. 创建socket（TCP）
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd < 0)
    {
        ESP_LOGE(TAG, "socket 创建失败");
        vTaskDelete(NULL);
        return;
    }

    // 2. 绑定端口
    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = htonl(INADDR_ANY);
    server_addr.sin_port = htons(TCP_SERVER_PORT);

    if (bind(server_fd, (struct sockaddr *)&server_addr, sizeof(server_addr)) < 0)
    {
        ESP_LOGE(TAG, "bind 端口 %d 失败", TCP_SERVER_PORT);
        close(server_fd);
        vTaskDelete(NULL);
        return;
    }

    // 3. 开启监听，等待客户端连接
    listen(server_fd, 1);
    ESP_LOGI(TAG, "TCP服务器已启动，端口：%d", TCP_SERVER_PORT);

    while (1)
    {
        // 4. 阻塞等待客户端接入
        client_fd = accept(server_fd, (struct sockaddr *)&client_addr, &client_len);
        if (client_fd >= 0)
        {
            ESP_LOGI(TAG, "客户端已接入: %s:%d",
                     inet_ntoa(client_addr.sin_addr), ntohs(client_addr.sin_port));
            while (1)
            {
                // 接收电脑发来的数据
                int len = recv(client_fd, rx_buffer, sizeof(rx_buffer) - 1, 0);
                if (len > 0)
                {
                    rx_buffer[len] = '\0';
                    ESP_LOGI(TAG, "收到客户端消息: %s", rx_buffer);
                    // 回发数据给电脑
                    send(client_fd, "OK\r\n", 4, 0);
                }
                else if (len == 0 || len < 0)
                {
                    ESP_LOGI(TAG, "客户端断开连接");
                    close(client_fd);
                    break;
                }
            }
        }
    }
    close(server_fd);
    vTaskDelete(NULL);
}

{
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());
    esp_netif_create_default_wifi_ap();

    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));

    ESP_ERROR_CHECK(esp_event_handler_instance_register(
        WIFI_EVENT,
        ESP_EVENT_ANY_ID,

        NULL,
        NULL));

    wifi_config_t wifi_config = {
        .ap = {
            .ssid = EXAMPLE_ESP_WIFI_SSID,
            .ssid_len = strlen(EXAMPLE_ESP_WIFI_SSID),
            .channel = EXAMPLE_ESP_WIFI_CHANNEL,
            .password = EXAMPLE_ESP_WIFI_PASS,
            .max_connection = EXAMPLE_MAX_STA_CONN,
            .authmode = WIFI_AUTH_WPA_WPA2_PSK},
    };
    if (strlen(EXAMPLE_ESP_WIFI_PASS) == 0)
    {
        wifi_config.ap.authmode = WIFI_AUTH_OPEN;
    }

    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_AP));
    ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_AP, &wifi_config));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_LOGI(TAG, "WiFi AP 启动完成");
    ESP_LOGI(TAG, "SSID: %s", EXAMPLE_ESP_WIFI_SSID);
    ESP_LOGI(TAG, "Password: %s", EXAMPLE_ESP_WIFI_PASS);

    xTaskCreate(udp_server_task, "udp_server", 4096, NULL, 5, NULL);
    // 创建TCP服务器任务
    xTaskCreate(tcp_server_task, "tcp_server", 4096, NULL, 5, NULL);
}
```

### 实验二　ICM20602 惯性数据采集实验

#### 实验目的

- 掌握 I2C 总线的初始化与设备挂载流程。
- 掌握依据数据手册读取寄存器的方法。
- 完成三轴加速度与三轴角速度的采集。

#### 实验原理

ICM20602 通过 I2C 提供寄存器访问接口。数据寄存器为高字节在前，需拼接为 16 位有符号数。原始值除以灵敏度即得到物理量：加速度除以 `LSB/g` 换算为 g，再乘以 9.8 换算为 m/s²；角速度除以 `LSB/(°/s)` 换算为 °/s，再乘以 π/180 换算为 rad/s。

#### 硬件说明

| 器件 | 引脚 | 说明 |
| --- | --- | --- |
| 板载 ICM20602 | I2C_SDA = GPIO41，I2C_SCL = GPIO42 | 已在板内连接，无需外部跳线 |

#### 参考代码

> **工程配置**：`main/CMakeLists.txt` 的 `REQUIRES` 需声明 `driver icm20602`。

**`main/main.c`**

```c
#include <stdio.h>
#include <stdint.h>
#include "esp_log.h"
#include "esp_err.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/i2c_master.h"
#include "icm20602.h"

static const char *TAG = "MAIN";

// I2C 引脚
#define I2C_SDA_GPIO 41
#define I2C_SCL_GPIO 42
#define I2C_FREQ_HZ 400000

// 测试打印周期
#define TEST_PERIOD_MS 100

void app_main(void)
{
    esp_err_t ret;

    //  1. 初始化 I2C 总线
    i2c_master_bus_config_t bus_cfg = {
        .clk_source = I2C_CLK_SRC_DEFAULT,
        .i2c_port = I2C_NUM_0,
        .scl_io_num = I2C_SCL_GPIO,
        .sda_io_num = I2C_SDA_GPIO,
        .glitch_ignore_cnt = 7,
        .flags.enable_internal_pullup = true,
    };
    i2c_master_bus_handle_t bus = NULL;
    ret = i2c_new_master_bus(&bus_cfg, &bus);
    if (ret != ESP_OK)
    {
        ESP_LOGE(TAG, "I2C bus init failed: %s", esp_err_to_name(ret));
        return;
    }
    ESP_LOGI(TAG, "I2C bus OK (SDA=%d, SCL=%d)", I2C_SDA_GPIO, I2C_SCL_GPIO);

    //  2. 挂载 ICM20602 设备
    i2c_device_config_t dev_cfg = {
        .dev_addr_length = I2C_ADDR_BIT_7,
        .device_address = ICM20602_ADDR_LOW, // 0x68, SA0 接地
        .scl_speed_hz = I2C_FREQ_HZ,
    };
    i2c_master_dev_handle_t i2c_dev = NULL;
    ret = i2c_master_bus_add_device(bus, &dev_cfg, &i2c_dev);
    if (ret != ESP_OK)
    {
        ESP_LOGE(TAG, "Add ICM20602 device failed: %s", esp_err_to_name(ret));
        return;
    }

    //  3. 初始化传感器（内部会做 WHO_AM_I 校验 + 软复位 + 量程配置）
    icm20602_config_t cfg;
    icm20602_init_default_config(&cfg);
    // 可按需修改量程，例如：
    // cfg.acc_range  = ICM20602_ACC_2G;
    // cfg.gyro_range = ICM20602_GYRO_1000DPS;

    icm20602_t imu;
    ret = icm20602_init(&imu, i2c_dev, &cfg);
    if (ret != ESP_OK)
    {
        ESP_LOGE(TAG, "ICM20602 init failed: %s", esp_err_to_name(ret));
        return;
    }
    ESP_LOGI(TAG, "ICM20602 init OK");

    //  4. 循环读取并打印
    icm20602_raw_data_t raw;
    icm20602_data_t data;
    uint32_t sample = 0;

    while (1)
    {
        ret = icm20602_read_raw(&imu, &raw);
        if (ret != ESP_OK)
        {
            ESP_LOGE(TAG, "read raw failed: %s", esp_err_to_name(ret));
        }
        else
        {
            ESP_LOGI(TAG,
                     "[%lu] RAW  ax=%6d ay=%6d az=%6d | gx=%6d gy=%6d gz=%6d | temp=%6d",
                     (unsigned long)sample,
                     raw.acc_x, raw.acc_y, raw.acc_z,
                     raw.gyro_x, raw.gyro_y, raw.gyro_z,
                     raw.temp);
        }

        ret = icm20602_read_data(&imu, &data);
        if (ret != ESP_OK)
        {
            ESP_LOGE(TAG, "read data failed: %s", esp_err_to_name(ret));
        }
        else
        {
            ESP_LOGI(TAG,
                     "[%lu] PHY  ax=%+.3fg ay=%+.3fg az=%+.3fg | "
                     "gx=%+7.2f gy=%+7.2f gz=%+7.2f dps | temp=%+.1f C",
                     (unsigned long)sample,
                     data.acc_x_g, data.acc_y_g, data.acc_z_g,
                     data.gyro_x_dps, data.gyro_y_dps, data.gyro_z_dps,
                     data.temp_c);
        }

        sample++;
        vTaskDelay(pdMS_TO_TICKS(TEST_PERIOD_MS));
    }
}
```

```c
// I2C 地址

// WHO_AM_I 期望值
#define ICM20602_WHO_AM_I_REG 0x75

    int16_t acc_x;
    int16_t acc_y;
    int16_t acc_z;
    int16_t temp;
    int16_t gyro_x;
    int16_t gyro_y;
    int16_t gyro_z;

    float acc_x_g;

    float gyro_x_dps;
    float gyro_y_dps;
    float gyro_z_dps;
```

**`components/icm20602/icm20602.c`**

```c
#include "icm20602.h"
#include <string.h>
#include <math.h>
#include "esp_log.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

static const char *TAG = "ICM20602";

// 寄存器
#define REG_SMPLRT_DIV 0x19
#define REG_CONFIG 0x1A
#define REG_GYRO_CONFIG 0x1B
#define REG_ACCEL_CONFIG 0x1C
#define REG_ACCEL_CONFIG2 0x1D
#define REG_ACCEL_XOUT_H 0x3B
#define REG_PWR_MGMT_1 0x6B
#define REG_PWR_MGMT_2 0x6C

// I2C 底层
static esp_err_t write_reg(icm20602_t *dev, uint8_t reg, uint8_t value)
{
    uint8_t buf[2] = {reg, value};
    return i2c_master_transmit(dev->i2c_dev, buf, 2, 100);
}

static esp_err_t read_regs(icm20602_t *dev, uint8_t reg, uint8_t *data, size_t len)
{
    return i2c_master_transmit_receive(dev->i2c_dev, &reg, 1, data, len, 100);
}

static int16_t make_int16(uint8_t hi, uint8_t lo)
{
    return (int16_t)(((uint16_t)hi << 8) | lo);
}

// 量程
static float acc_scale_from_range(icm20602_acc_range_t r)
{
    switch (r)
    {
    case ICM20602_ACC_2G:
        return 16384.0f;
    case ICM20602_ACC_4G:
        return 8192.0f;
    case ICM20602_ACC_8G:
        return 4096.0f;
    case ICM20602_ACC_16G:
        return 2048.0f;
    default:
        return 4096.0f;
    }
}

static float gyro_scale_from_range(icm20602_gyro_range_t r)
{
    switch (r)
    {
    case ICM20602_GYRO_250DPS:
        return 131.0f;
    case ICM20602_GYRO_500DPS:
        return 65.5f;
    case ICM20602_GYRO_1000DPS:
        return 32.8f;
    case ICM20602_GYRO_2000DPS:
        return 16.4f;
    default:
        return 16.4f;
    }
}

// 公开 API
void icm20602_init_default_config(icm20602_config_t *config)
{
    config->i2c_address = ICM20602_ADDR_LOW;
    config->acc_range = ICM20602_ACC_8G;
    config->gyro_range = ICM20602_GYRO_2000DPS;
    config->sample_rate_div = 9; // ~100Hz
}

esp_err_t icm20602_init(icm20602_t *dev, i2c_master_dev_handle_t i2c_dev,
                        const icm20602_config_t *config)
{
    if (!dev || !i2c_dev || !config)
        return ESP_ERR_INVALID_ARG;

    memset(dev, 0, sizeof(*dev));
    dev->i2c_dev = i2c_dev;
    dev->i2c_address = config->i2c_address;

    vTaskDelay(pdMS_TO_TICKS(100));

    //  WHO_AM_I 检查
    uint8_t who = 0;
    esp_err_t ret = read_regs(dev, ICM20602_WHO_AM_I_REG, &who, 1);
    if (ret != ESP_OK)
    {
        ESP_LOGE(TAG, "WHO_AM_I read failed: %s", esp_err_to_name(ret));
        return ret;
    }
    ESP_LOGI(TAG, "WHO_AM_I = 0x%02X (expect 0x%02X)", who, ICM20602_WHO_AM_I_VAL);

    if (who != ICM20602_WHO_AM_I_VAL)
    {
        ESP_LOGE(TAG, "WHO_AM_I mismatch! Check wiring/address.");
        return ESP_ERR_NOT_FOUND;
    }

    //  软件复位
    ret = write_reg(dev, REG_PWR_MGMT_1, 0x80);
    if (ret != ESP_OK)
        return ret;
    vTaskDelay(pdMS_TO_TICKS(100));

    //  退出睡眠 + 时钟源 = PLL
    ret = write_reg(dev, REG_PWR_MGMT_1, 0x01);
    if (ret != ESP_OK)
        return ret;
    vTaskDelay(pdMS_TO_TICKS(50));

    //  开启加速度计 XYZ + 陀螺仪 XYZ
    ret = write_reg(dev, REG_PWR_MGMT_2, 0x00);
    if (ret != ESP_OK)
        return ret;

    //  采样率分频
    write_reg(dev, REG_SMPLRT_DIV, config->sample_rate_div);

    //  低通滤波
    write_reg(dev, REG_CONFIG, 0x03);

    //  陀螺仪量程
    write_reg(dev, REG_GYRO_CONFIG, (uint8_t)config->gyro_range);

    //  加速度计量程
    write_reg(dev, REG_ACCEL_CONFIG, (uint8_t)config->acc_range);

    //  加速度计低通滤波
    write_reg(dev, REG_ACCEL_CONFIG2, 0x03);

    //  计算比例因子
    dev->acc_scale = acc_scale_from_range(config->acc_range);
    dev->gyro_scale = gyro_scale_from_range(config->gyro_range);

    dev->initialized = true;
    ESP_LOGI(TAG, "Init OK (acc=%.0f LSB/g, gyro=%.1f LSB/dps)",
             dev->acc_scale, dev->gyro_scale);
    return ESP_OK;
}

{
    if (!dev || !dev->initialized || !raw)
        return ESP_ERR_INVALID_ARG;

    uint8_t buf[14] = {0};

        return ret;
    }

    raw->acc_x = make_int16(buf[0], buf[1]);
    raw->acc_y = make_int16(buf[2], buf[3]);
    raw->acc_z = make_int16(buf[4], buf[5]);
    raw->temp = make_int16(buf[6], buf[7]);
    raw->gyro_x = make_int16(buf[8], buf[9]);
    raw->gyro_y = make_int16(buf[10], buf[11]);
    raw->gyro_z = make_int16(buf[12], buf[13]);

    return ESP_OK;
}

esp_err_t icm20602_read_data(icm20602_t *dev, icm20602_data_t *data)
{
    icm20602_raw_data_t raw;
    esp_err_t ret = icm20602_read_raw(dev, &raw);
    if (ret != ESP_OK)
        return ret;

    data->acc_x_g = raw.acc_x / dev->acc_scale;
    data->acc_y_g = raw.acc_y / dev->acc_scale;
    data->acc_z_g = raw.acc_z / dev->acc_scale;

    data->gyro_y_dps = raw.gyro_y / dev->gyro_scale;
    data->gyro_z_dps = raw.gyro_z / dev->gyro_scale;
    data->temp_c = raw.temp / 326.8f + 25.0f;

    return ESP_OK;
}
```

### 实验三　PMW3901 光流传感器 SPI 采集实验

#### 实验目的

- 掌握 SPI 总线的初始化与设备挂载流程。
- 完成光流传感器运动增量（dx / dy）的读取。
- 建立光流测位移与惯性测角度的配合关系。

#### 实验原理

PMW3901 通过拍摄地面纹理并比较相邻帧的位移计算运动增量。SPI 通信需正确配置时钟极性与相位（模式 0）。读取寄存器时需将地址最高位置 1 表示读操作。传感器输出的 dx/dy 为相对位移增量，需结合高度信息才能换算为实际地面位移。

```c
esp_err_t pmw3901_init(spi_device_handle_t spi_dev)
{
    if (g_initialized)
        return ESP_OK;
    if (!spi_dev)
        return ESP_ERR_INVALID_ARG;

    g_spi_dev = spi_dev;

    //  SPI 总线复位时序 (原库: CS toggle)
    {
        uint8_t dummy = 0;
        spi_transaction_t trans = {
            .length = 8,
            .tx_buffer = &dummy,
        };
        spi_device_transmit(g_spi_dev, &trans);
        vTaskDelay(pdMS_TO_TICKS(2));
    }

    //  上电复位
    esp_err_t ret = register_write(0x3A, 0x5A);
    if (ret != ESP_OK)
    {
        ESP_LOGE(TAG, "Power-on reset write failed");
        g_spi_dev = NULL;
        return ret;
    }
    vTaskDelay(pdMS_TO_TICKS(5));

    //  芯片 ID 检查
    uint8_t chip_id = register_read(0x00);
    uint8_t inv_id = register_read(0x5F);

    ESP_LOGI(TAG, "Chip ID: 0x%02X (expect 0x49), Inverse: 0x%02X (~0x49=0xB6)",
             chip_id, inv_id);

    if (chip_id != 0x49 && inv_id != 0xB8)
    {
        ESP_LOGE(TAG, "Chip ID mismatch! Check wiring.");
        g_spi_dev = NULL;
        return ESP_ERR_NOT_FOUND;
    }

    //  预读 Motion 寄存器 (清空残留数据)
    register_read(0x02);
    register_read(0x03);
    register_read(0x04);
    register_read(0x05);
    register_read(0x06);
    vTaskDelay(pdMS_TO_TICKS(1));

    //  性能优化寄存器
    init_registers();

    g_initialized = true;
    ESP_LOGI(TAG, "PMW3901 init OK");
    return ESP_OK;
}

esp_err_t pmw3901_read_motion(int16_t *dx, int16_t *dy)
{
    if (!g_initialized || !dx || !dy)
        return ESP_ERR_INVALID_STATE;

    // 读 0x02 触发 motion 寄存器锁存 + 清零
    register_read(0x02);

    // 读 Delta_X (0x03 low, 0x04 high)
    uint8_t xh = register_read(0x04);
    uint8_t xl = register_read(0x03);
    *dx = (int16_t)(((uint16_t)xh << 8) | xl);

    // 读 Delta_Y (0x05 low, 0x06 high)
    uint8_t yh = register_read(0x06);
    uint8_t yl = register_read(0x05);
    *dy = (int16_t)(((uint16_t)yh << 8) | yl);

    return ESP_OK;
}

esp_err_t pmw3901_read_squal(uint8_t *squal)
{
    if (!g_initialized || !squal)
        return ESP_ERR_INVALID_STATE;

    // 0x55 = Surface_Quality (SQUAL), 图像质量 0~255
    *squal = register_read(0x07);
    return ESP_OK;
}

{

        return ESP_ERR_INVALID_STATE;

    register_write(0x7F, 0x07);
    register_write(0x41, 0x1D);
    register_write(0x4C, 0x00);
    register_write(0x7F, 0x08);
    register_write(0x6A, 0x38);
    register_write(0x7F, 0x00);
    register_write(0x55, 0x04);
    register_write(0x40, 0x80);
    register_write(0x4D, 0x11);
    register_write(0x70, 0x00);
    register_write(0x58, 0xFF);

    // 等待帧捕获就绪
    int check;
    do
    {
        check = register_read(0x58) >> 6;
    } while (check == 0x03);

    ets_delay_us(50);
    return ESP_OK;
}

esp_err_t pmw3901_read_frame_buffer(uint8_t *buffer)
{
    if (!g_initialized || !buffer)
        return ESP_ERR_INVALID_STATE;

    int count = 0;
    uint8_t mask = 0x0C;

    for (int i = 0; i < 1225; i++)
    {
        uint8_t a, hold;
        // 等待有效数据
        do
        {
            a = register_read(0x58);
            hold = a >> 6;
        } while (hold == 0x00 || hold == 0x03);

        if (hold == 0x01)
        {
            uint8_t b = register_read(0x58);
            uint8_t pixel = (uint8_t)(a << 2) | (b & mask);
            buffer[count++] = pixel;
        }
    }

    // 恢复并等待下一次就绪
    register_write(0x70, 0x00);
    register_write(0x58, 0xFF);

    int check;
    do
    {
        check = register_read(0x58) >> 6;
    } while (check == 0x03);

    return ESP_OK;
}
```

### 实验五　SPL06-001 气压计 SPI 采集实验

#### 实验目的

- 掌握 SPI 传感器寄存器的读写时序。
- 理解气压与海拔的换算关系。
- 完成气压与温度数据的采集与高度估算。

#### 实验原理

SPL06-001 输出气压与温度原始值，需结合出厂标定系数进行温度补偿，再换算出真实气压值。气压随高度升高而降低，国际标准大气模型下可用下式估算相对高度：

```
高度差(m) ≈ 44330 × [1 − (P / P0) ^ 0.1903]
```

其中 P 为当前气压，P0 为基准气压。该公式为近似模型，实际应用中需以起飞点的气压作为基准值。

```c
// SPL06 气压计 — SPI 驱动

#include "spl06001.h"

#include <string.h>
#include <math.h>

#include "esp_log.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/spi_master.h"

static const char *TAG = "SPL06";

#define SPL06_REG_PRS_B2 0x00
#define SPL06_REG_TMP_B2 0x03
#define SPL06_REG_PRS_CFG 0x06
#define SPL06_REG_TMP_CFG 0x07
#define SPL06_REG_MEAS_CFG 0x08
#define SPL06_REG_CFG_REG 0x09
#define SPL06_REG_RESET 0x0C
#define SPL06_REG_ID 0x0D
#define SPL06_REG_COEF 0x10

#define SPL06_ID_EXPECTED 0x10
#define SPL06_PRS_CFG_VALUE   0x03
#define SPL06_TMP_CFG_VALUE   0x83
#define SPL06_MEAS_CFG_VALUE  0x07

#define SPL06_SCALE_FACTOR_8X 7864320.0f

static spi_device_handle_t spl06_spi = NULL;

typedef struct {
    int16_t c0;
    int16_t c1;
    int32_t c00;
    int32_t c10;
    int16_t c01;
    int16_t c11;
    int16_t c20;
    int16_t c21;
    int16_t c30;
} spl06_coef_t;

static spl06_coef_t g_coef;

static esp_err_t spl06_write_reg(uint8_t reg, uint8_t value)
{
    uint8_t tx_data[2] = { reg & 0x7F, value };
    spi_transaction_t trans = {
        .length    = 8 * sizeof(tx_data),
        .tx_buffer = tx_data,
    };
    return spi_device_transmit(spl06_spi, &trans);
}

static esp_err_t spl06_read_regs(uint8_t reg, uint8_t *data, uint8_t len)
{
    if (len == 0 || len > 32) return ESP_ERR_INVALID_ARG;

    uint8_t tx_data[33] = {0};
    uint8_t rx_data[33] = {0};
    tx_data[0] = reg | 0x80;

    spi_transaction_t trans = {
        .length    = 8 * (len + 1),
        .tx_buffer = tx_data,
        .rx_buffer = rx_data,
    };

    esp_err_t ret = spi_device_transmit(spl06_spi, &trans);
    if (ret != ESP_OK) return ret;

    memcpy(data, &rx_data[1], len);
    return ESP_OK;
}

static esp_err_t spl06_read_reg(uint8_t reg, uint8_t *value)
{
    return spl06_read_regs(reg, value, 1);
}

static int32_t sign_extend(uint32_t raw, uint8_t bits)
{
    uint32_t sign_bit = 1U << (bits - 1);
    if (raw & sign_bit) raw |= ~((1U << bits) - 1);
    return (int32_t)raw;
}

static esp_err_t spl06_read_raw24(uint8_t start_reg, int32_t *raw)
{
    uint8_t buf[3] = {0};
    esp_err_t ret = spl06_read_regs(start_reg, buf, 3);
    if (ret != ESP_OK) return ret;

    uint32_t value = ((uint32_t)buf[0] << 16) | ((uint32_t)buf[1] << 8) | buf[2];
    *raw = sign_extend(value, 24);
    return ESP_OK;
}

static esp_err_t spl06_read_calibration_data(void)
{
    uint8_t coef[18] = {0};
    esp_err_t ret = spl06_read_regs(SPL06_REG_COEF, coef, 18);
    if (ret != ESP_OK) return ret;

    g_coef.c0  = sign_extend(((uint32_t)coef[0] << 4) | ((uint32_t)coef[1] >> 4), 12);
    g_coef.c1  = sign_extend((((uint32_t)coef[1] & 0x0F) << 8) | coef[2], 12);
    g_coef.c00 = sign_extend(((uint32_t)coef[3] << 12) | ((uint32_t)coef[4] << 4) | ((uint32_t)coef[5] >> 4), 20);
    g_coef.c10 = sign_extend((((uint32_t)coef[5] & 0x0F) << 16) | ((uint32_t)coef[6] << 8) | coef[7], 20);
    g_coef.c01 = sign_extend(((uint32_t)coef[8] << 8) | coef[9], 16);
    g_coef.c11 = sign_extend(((uint32_t)coef[10] << 8) | coef[11], 16);
    g_coef.c20 = sign_extend(((uint32_t)coef[12] << 8) | coef[13], 16);
    g_coef.c21 = sign_extend(((uint32_t)coef[14] << 8) | coef[15], 16);
    g_coef.c30 = sign_extend(((uint32_t)coef[16] << 8) | coef[17], 16);

    ESP_LOGI(TAG, "Calibration data loaded");
    ESP_LOGI(TAG, "c0=%d, c1=%d, c00=%ld, c10=%ld",
             g_coef.c0, g_coef.c1, g_coef.c00, g_coef.c10);
    return ESP_OK;
}

static esp_err_t spl06_wait_ready(void)
{
    for (int i = 0; i < 50; i++) {
        uint8_t status = 0;
        esp_err_t ret = spl06_read_reg(SPL06_REG_MEAS_CFG, &status);
        if (ret != ESP_OK) return ret;

        if ((status & 0xC0) == 0xC0) {
            ESP_LOGI(TAG, "Sensor ready, MEAS_CFG=0x%02X", status);
            return ESP_OK;
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
    ESP_LOGE(TAG, "Sensor ready timeout");
    return ESP_ERR_TIMEOUT;
}

static esp_err_t spl06_wait_data_ready(void)
{
    for (int i = 0; i < 50; i++) {
        uint8_t status = 0;
        esp_err_t ret = spl06_read_reg(SPL06_REG_MEAS_CFG, &status);
        if (ret != ESP_OK) return ret;

        if ((status & 0x30) == 0x30) return ESP_OK;
        vTaskDelay(pdMS_TO_TICKS(10));
    }
    ESP_LOGW(TAG, "Data ready timeout");
    return ESP_ERR_TIMEOUT;
}

esp_err_t spl06_init(spi_device_handle_t spi_dev)
{
    if (!spi_dev) return ESP_ERR_INVALID_ARG;
    spl06_spi = spi_dev;

    esp_err_t ret;

    uint8_t id = 0;
    ret = spl06_read_reg(SPL06_REG_ID, &id);
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "Read ID failed: %s", esp_err_to_name(ret));
        return ret;
    }
    ESP_LOGI(TAG, "SPL06 ID = 0x%02X", id);

    if (id != SPL06_ID_EXPECTED) {
        ESP_LOGW(TAG, "ID mismatch: got 0x%02X, expected 0x10. Will try to use anyway.", id);
    }

    ret = spl06_write_reg(SPL06_REG_RESET, 0x09);
    if (ret != ESP_OK) return ret;
    vTaskDelay(pdMS_TO_TICKS(50));

    ret = spl06_wait_ready();
    if (ret != ESP_OK) return ret;

    ret = spl06_read_calibration_data();
    if (ret != ESP_OK) {
        ESP_LOGE(TAG, "Read calibration data failed");
        return ret;
    }

    spl06_write_reg(SPL06_REG_PRS_CFG, SPL06_PRS_CFG_VALUE);
    spl06_write_reg(SPL06_REG_TMP_CFG, SPL06_TMP_CFG_VALUE);
    spl06_write_reg(SPL06_REG_CFG_REG, 0x00);
    spl06_write_reg(SPL06_REG_MEAS_CFG, SPL06_MEAS_CFG_VALUE);
    vTaskDelay(pdMS_TO_TICKS(100));

    ESP_LOGI(TAG, "SPL06 init done");
    return ESP_OK;
}

esp_err_t spl06_read(spl06_data_t *data)
{
    if (data == NULL) return ESP_ERR_INVALID_ARG;

    esp_err_t ret = spl06_wait_data_ready();
    if (ret != ESP_OK) return ret;

    int32_t raw_temp = 0, raw_pressure = 0;
    ret = spl06_read_raw24(SPL06_REG_TMP_B2, &raw_temp);
    if (ret != ESP_OK) return ret;
    ret = spl06_read_raw24(SPL06_REG_PRS_B2, &raw_pressure);
    if (ret != ESP_OK) return ret;

    float Traw_sc = raw_temp / SPL06_SCALE_FACTOR_8X;
    float Praw_sc = raw_pressure / SPL06_SCALE_FACTOR_8X;

    float temperature_c = g_coef.c0 * 0.5f + g_coef.c1 * Traw_sc;
    float pressure_pa =
        g_coef.c00 +
        Praw_sc * (g_coef.c10 + Praw_sc * (g_coef.c20 + Praw_sc * g_coef.c30)) +
        Traw_sc * g_coef.c01 +
        Traw_sc * Praw_sc * (g_coef.c11 + Praw_sc * g_coef.c21);

    data->raw_temperature = raw_temp;
    data->raw_pressure    = raw_pressure;
    data->temperature_c   = temperature_c;
    data->pressure_pa     = pressure_pa;
    data->pressure_hpa    = pressure_pa / 100.0f;
    data->altitude_m      = spl06_pressure_to_altitude(pressure_pa);

    return ESP_OK;
}

float spl06_pressure_to_altitude(float pressure_pa)
{
    return 44330.0f * (1.0f - powf(pressure_pa / 101325.0f, 0.19029495f));
}
```

### 实验六　VL53L0X 激光测距 I2C 采集实验

#### 实验目的

- 掌握复杂 I2C 传感器的初始化与配置流程。
- 理解测量模式与时序预算对精度的影响。
- 完成单次测量与连续测量两种采集方式。

#### 实验原理

VL53L0X 采用 ToF（飞行时间）原理测距。初始化需要依次完成寄存器配置、参考校准与 SPAD 校准。时序预算（timing budget）决定单次测量允许的最长时间，预算越大精度与量程越好，但测量周期越长。器件支持单次测量与连续测量两种模式，也可通过中断引脚在数据就绪时触发读取。

```c
    i2c_master_dev_handle_t vl53l0x_i2c_dev = NULL;
    ret = i2c_master_bus_add_device(i2c_bus_handle, &VL53L0X_dev_cfg, &vl53l0x_i2c_dev);
    if (ret != ESP_OK)
    {
        ESP_LOGE(TAG, "I2C device add failed: %s", esp_err_to_name(ret));
        return;
    }
    ESP_LOGI(TAG, "I2C device OK (SDA=%d, SCL=%d)", I2C_SDA_GPIO, I2C_SCL_GPIO);

    vl53l0x_config_t tof_config;
    vl53l0x_init_default_config(&tof_config);
    tof_config.i2c_address = VL53L0X_ADDR;
    tof_config.timeout_ms = 500;
    tof_config.io_2v8 = true;
    tof_config.interrupt_pin = 1;

    vl53l0x_t tof_dev;
    ret = vl53l0x_init(&tof_dev, vl53l0x_i2c_dev, &tof_config);
    if (ret != ESP_OK)
    {
        ESP_LOGE(TAG, "VL53L0X init failed: %s", esp_err_to_name(ret));
    }
    else
    {
        vl53l0x_set_range_mode(&tof_dev, 2);
        ESP_LOGI(TAG, "VL53L0X OK (long-range)");
    }

    bool tof_ok = (ret == ESP_OK);
    uint16_t dist = 0;
    bool dist_ok = false;

    while (1)
    {
        vTaskDelay(1000 / portTICK_PERIOD_MS);
```

### 实验七　上位机解析陀螺仪数据计算偏航角

#### 实验目的

- 掌握将传感器数据通过串口或网络上报到上位机的方法。
- 理解角速度积分得到角度的计算过程。
- 分析积分漂移产生的原因与补偿思路。

#### 实验原理

角速度对时间积分即得到角度：

```
偏航角(rad) = Σ (gz(rad/s) × dt)
```

由于陀螺仪存在零偏与随机噪声，长时间积分会产生累积漂移。工程上采用两种方法抑制：一是测量前进行零偏校准，将静止状态下的多次采样平均值作为零偏并从原始数据中扣除；二是与加速度计或磁力计进行互补滤波或卡尔曼滤波融合。

#### 硬件说明

板载 ICM20602 已与 ESP32-S3 内部连接（SCL = GPIO42，SDA = GPIO41），无需额外接线。

```cpp
#include "GyroParser.h"
#include <QRegularExpression>
#include <QDebug>
#include <QtMath>

static double wrap180(double a)
{
    while (a > 180.0)
        a -= 360.0;
    while (a < -180.0)
        a += 360.0;
    return a;
}

static double wrap360(double a)
{
    while (a >= 360.0)
        a -= 360.0;
    while (a < 0.0)
        a += 360.0;
    return a;
}

GyroParser::GyroParser(QObject *parent)
    : QObject(parent)
{
}

void GyroParser::processLine(const QString &line)
{
    if (!line.contains(QStringLiteral("PHY")))
        return;

    static const QRegularExpression tsRe(QStringLiteral(R"(\(\s*(\d+)\s*\))"));

    static const QRegularExpression axRe(QStringLiteral(R"(ax\s*=\s*([-+]?\d+(?:\.\d+)?))"));
    static const QRegularExpression ayRe(QStringLiteral(R"(ay\s*=\s*([-+]?\d+(?:\.\d+)?))"));
    static const QRegularExpression azRe(QStringLiteral(R"(az\s*=\s*([-+]?\d+(?:\.\d+)?))"));

    static const QRegularExpression gxRe(QStringLiteral(R"(gx\s*=\s*([-+]?\d+(?:\.\d+)?))"));
    static const QRegularExpression gyRe(QStringLiteral(R"(gy\s*=\s*([-+]?\d+(?:\.\d+)?))"));
    static const QRegularExpression gzRe(QStringLiteral(R"(gz\s*=\s*([-+]?\d+(?:\.\d+)?))"));

    const auto tsM = tsRe.match(line);
    const auto axM = axRe.match(line);
    const auto ayM = ayRe.match(line);
    const auto azM = azRe.match(line);
    const auto gxM = gxRe.match(line);
    const auto gyM = gyRe.match(line);
    const auto gzM = gzRe.match(line);

    if (!tsM.hasMatch() || !axM.hasMatch() || !ayM.hasMatch() || !azM.hasMatch() ||
        !gxM.hasMatch() || !gyM.hasMatch() || !gzM.hasMatch())
        return;

    const qint64 ts = tsM.captured(1).toLongLong();
    const double ax = axM.captured(1).toDouble();
    const double ay = ayM.captured(1).toDouble();
    const double az = azM.captured(1).toDouble();
    const double gx = gxM.captured(1).toDouble();
    const double gy = gyM.captured(1).toDouble();
    const double gz = gzM.captured(1).toDouble();

    // 标定阶段
    if (m_calibrating)
    {
        m_calibX.append(gx);
        m_calibY.append(gy);
        m_calibZ.append(gz);
        if (m_calibZ.size() >= m_calibTarget)
            finishCalibration();
        return;
    }

    if (m_lastTs < 0)
    {
        m_lastTs = ts;
        return;
    }

    double dt = (ts - m_lastTs) / 1000.0; // 毫秒 -> 秒
    m_lastTs = ts;
    if (dt <= 0.0 || dt > 5.0)
        return;

    const double rollAcc = qRadiansToDegrees(qAtan2(ay, az));
    const double pitchAcc = qRadiansToDegrees(qAtan2(-ax, qSqrt(ay * ay + az * az)));

    double k = dt / (dt + m_tau);

    const double mag = qSqrt(ax * ax + ay * ay + az * az);
    if (mag < 0.5 || mag > 2.0)
        k = 0.0;

    // 横滚/俯仰
    m_roll = (1.0 - k) * (m_roll + (gx - m_biasX) * dt) + k * rollAcc;
    m_pitch = (1.0 - k) * (m_pitch + (gy - m_biasY) * dt) + k * pitchAcc;

    // 偏航
    m_yaw += (gz - m_biasZ) * dt;

    m_roll = wrap180(m_roll);
    m_pitch = wrap180(m_pitch);
    m_yaw = wrap360(m_yaw);

    emit attitudeChanged(m_roll, m_pitch, m_yaw);
}

void GyroParser::setGyroBias(double gxDps, double gyDps, double gzDps)
{
    m_biasX = gxDps;
    m_biasY = gyDps;
    m_biasZ = gzDps;
    m_calibrating = false;
    m_calibX.clear();
    m_calibY.clear();
    m_calibZ.clear();
    m_roll = m_pitch = m_yaw = 0.0;
    m_lastTs = -1;
    emit calibrationDone(m_biasX, m_biasY, m_biasZ);
}

void GyroParser::startCalibration(int sampleCount)
{
    m_calibTarget = sampleCount > 0 ? sampleCount : 50;
    m_calibX.clear();
    m_calibY.clear();
    m_calibZ.clear();
    m_calibrating = true;
    m_roll = m_pitch = m_yaw = 0.0;
    m_lastTs = -1;
}

void GyroParser::finishCalibration()
{
    if (m_calibX.isEmpty())
        return;

    auto avg = [](const QVector<double> &v)
    {
        double sum = 0.0;
        for (double x : v)
            sum += x;
        return sum / v.size();
    };

    m_biasX = avg(m_calibX);
    m_biasY = avg(m_calibY);
    m_biasZ = avg(m_calibZ);

    m_calibX.clear();
    m_calibY.clear();
    m_calibZ.clear();
    m_calibrating = false;
    m_roll = m_pitch = m_yaw = 0.0;
    m_lastTs = -1;

    emit calibrationDone(m_biasX, m_biasY, m_biasZ);
}

void GyroParser::reset()
{
    m_roll = m_pitch = m_yaw = 0.0;
    m_lastTs = -1;
    m_calibX.clear();
    m_calibY.clear();
    m_calibZ.clear();
    m_calibrating = true;
}
```

```cpp
QStringList SerialPortManager::getAvailablePorts() const {

	foreach(const QSerialPortInfo & info, QSerialPortInfo::availablePorts()) {
		portList << info.portName();
	}
	return portList;

void SerialPortManager::handleReadyRead() {
    m_recvBuf.append(serialPort->readAll()); // 追加到缓冲区

    // 循环查找换行符 \n，分割完整行
    while(true)
    {
        int idx = m_recvBuf.indexOf('\n');
        if(idx < 0) break; // 没有完整一行，退出，等待下次数据

        // 取出一行（包含 \r）
        QByteArray line = m_recvBuf.left(idx+1);
        m_recvBuf = m_recvBuf.mid(idx+1); // 缓冲区剩余数据保留

        line = line.trimmed(); // 去掉 \r \n 回车换行
        if(!line.isEmpty())
        {
            emit dataReceived(line); // 发出**完整一行**
        }
    }
}
```

### 实验八　上位机解析光流数据计算相对位移

#### 实验目的

- 掌握光流增量数据的上报与解析方法。
- 理解光流位移与高度之间的关系。
- 实现水平方向相对位移的估算。

#### 实验原理

光流传感器输出的 dx / dy 为图像平面上的像素位移增量，与实际地面位移的关系为：

```
实际位移 ≈ 像素增量 × 高度 ÷ 焦距系数
```

即同一像素位移对应的实际位移随高度线性增大。因此在定高飞行时，光流可用于水平位置估计；单独使用光流无法得到绝对位置，需要与惯性测量或视觉数据融合。

```c
// 公开 API
esp_err_t pmw3901_init(spi_device_handle_t spi_dev)
{
    if (g_initialized)
        return ESP_OK;
    if (!spi_dev)
        return ESP_ERR_INVALID_ARG;

    g_spi_dev = spi_dev;

    // ---- SPI 总线复位时序 (原库: CS toggle) ----
    {
        uint8_t dummy = 0;
        spi_transaction_t trans = {
            .length = 8,
            .tx_buffer = &dummy,
        };
        spi_device_transmit(g_spi_dev, &trans);
        vTaskDelay(pdMS_TO_TICKS(2));
    }

    // ---- 上电复位 ----
    esp_err_t ret = register_write(0x3A, 0x5A);
    if (ret != ESP_OK)
    {
        ESP_LOGE(TAG, "Power-on reset write failed");
        g_spi_dev = NULL;
        return ret;
    }
    vTaskDelay(pdMS_TO_TICKS(5));

    // ---- 芯片 ID 检查 ----
    uint8_t chip_id = register_read(0x00);
    uint8_t inv_id = register_read(0x5F);

    ESP_LOGI(TAG, "Chip ID: 0x%02X (expect 0x49), Inverse: 0x%02X (~0x49=0xB6)",
             chip_id, inv_id);

    if (chip_id != 0x49 && inv_id != 0xB8)
    {
        ESP_LOGE(TAG, "Chip ID mismatch! Check wiring.");
        g_spi_dev = NULL;
        return ESP_ERR_NOT_FOUND;
    }

    // ---- 预读 Motion 寄存器 (清空残留数据) ----
    register_read(0x02);
    register_read(0x03);
    register_read(0x04);
    register_read(0x05);
    register_read(0x06);
    vTaskDelay(pdMS_TO_TICKS(1));

    // ---- 性能优化寄存器 ----
    init_registers();

    g_initialized = true;
    ESP_LOGI(TAG, "PMW3901 init OK");
    return ESP_OK;
}

esp_err_t pmw3901_read_motion(int16_t *dx, int16_t *dy)
{
    if (!g_initialized || !dx || !dy)
        return ESP_ERR_INVALID_STATE;

    // 读 0x02 触发 motion 寄存器锁存 + 清零
    register_read(0x02);

    // 读 Delta_X (0x03 low, 0x04 high)
    uint8_t xh = register_read(0x04);
    uint8_t xl = register_read(0x03);
    *dx = (int16_t)(((uint16_t)xh << 8) | xl);

    // 读 Delta_Y (0x05 low, 0x06 high)
    uint8_t yh = register_read(0x06);
    uint8_t yl = register_read(0x05);
    *dy = (int16_t)(((uint16_t)yh << 8) | yl);

    return ESP_OK;
}

esp_err_t pmw3901_read_squal(uint8_t *squal)
{
    if (!g_initialized || !squal)
        return ESP_ERR_INVALID_STATE;

    // 0x55 = Surface_Quality (SQUAL), 图像质量 0~255
    *squal = register_read(0x07);
    return ESP_OK;
}

esp_err_t pmw3901_set_led(bool led_on)
{
    if (!g_initialized)
        return ESP_ERR_INVALID_STATE;

    vTaskDelay(pdMS_TO_TICKS(200));
    register_write(0x7F, 0x14);
    register_write(0x6F, led_on ? 0x1C : 0x00);
    register_write(0x7F, 0x00);

    return ESP_OK;
}

esp_err_t pmw3901_enable_frame_buffer(void)
{
    if (!g_initialized)
        return ESP_ERR_INVALID_STATE;

    register_write(0x7F, 0x07);
    register_write(0x41, 0x1D);
    register_write(0x4C, 0x00);
    register_write(0x7F, 0x08);
    register_write(0x6A, 0x38);
    register_write(0x7F, 0x00);
    register_write(0x55, 0x04);
    register_write(0x40, 0x80);
    register_write(0x4D, 0x11);
    register_write(0x70, 0x00);
    register_write(0x58, 0xFF);

    // 等待帧捕获就绪
    int check;
    do
    {
        check = register_read(0x58) >> 6;
    } while (check == 0x03);

    ets_delay_us(50);
    return ESP_OK;
}

esp_err_t pmw3901_read_frame_buffer(uint8_t *buffer)
{
    if (!g_initialized || !buffer)
        return ESP_ERR_INVALID_STATE;

    int count = 0;
    uint8_t mask = 0x0C;

    for (int i = 0; i < 1225; i++)
    {
        uint8_t a, hold;
        // 等待有效数据
        do
        {
            a = register_read(0x58);
            hold = a >> 6;
        } while (hold == 0x00 || hold == 0x03);

        if (hold == 0x01)
        {
            uint8_t b = register_read(0x58);
            uint8_t pixel = (uint8_t)(a << 2) | (b & mask);
            buffer[count++] = pixel;
        }
    }

    // 恢复并等待下一次就绪
    register_write(0x70, 0x00);
    register_write(0x58, 0xFF);

    int check;
    do
    {
        check = register_read(0x58) >> 6;
    } while (check == 0x03);

    return ESP_OK;
}
```

```cpp
#include "FlowParser.h"
#include <QRegularExpression>

FlowParser::FlowParser(QObject *parent)
    : QObject(parent)
{
    m_timer.start();
}

void FlowParser::processLine(const QString &line)
{
    if (line.contains(QStringLiteral("squal")))
    {
        static const QRegularExpression squalRe(QStringLiteral(R"(squal\s*=\s*(\d+))"));
        const auto m = squalRe.match(line);
        if (!m.hasMatch())
            return;
        m_squal = m.captured(1).toInt();

        double vx = 0.0, vy = 0.0;

        if (m_hasPending)
        {
            if (m_squal >= m_minSqual)
            {
                const double dmmX = m_pendingDx * m_heightMm * kFlowScale;
                const double dmmY = m_pendingDy * m_heightMm * kFlowScale;

                const qint64 nowMs = m_timer.elapsed();
                if (m_hasLastTs)
                {
                    const double dtSec = (nowMs - m_lastTsMs) / 1000.0;
                    if (dtSec > 1e-3)
                    {
                        vx = dmmX / dtSec;
                        vy = dmmY / dtSec;
                    }
                }
                m_lastTsMs = nowMs;
                m_hasLastTs = true;

                m_xMm += dmmX;
                m_yMm += dmmY;
            }
            m_hasPending = false;
        }

        emit sampleReady(m_xMm, m_yMm, m_heightMm, vx, vy, m_squal);
        return;
    }

    if (line.contains(QStringLiteral("Distance"), Qt::CaseInsensitive))
    {
        static const QRegularExpression distRe(QStringLiteral(R"(Distance\s*:\s*([-+]?\d+(?:\.\d+)?))"),
                                               QRegularExpression::CaseInsensitiveOption);
        const auto m = distRe.match(line);
        if (m.hasMatch())
            m_heightMm = m.captured(1).toDouble();
        return;
    }

    static const QRegularExpression dxRe(QStringLiteral(R"(dx\s*=\s*([-+]?\d+))"));
    static const QRegularExpression dyRe(QStringLiteral(R"(dy\s*=\s*([-+]?\d+))"));
    const auto dxM = dxRe.match(line);
    const auto dyM = dyRe.match(line);
    if (!dxM.hasMatch() || !dyM.hasMatch())
        return;

    m_pendingDx = dxM.captured(1).toDouble();
    m_pendingDy = dyM.captured(1).toDouble();
    m_hasPending = true;
}

void FlowParser::reset()
{
    m_xMm = m_yMm = 0.0;
    m_hasPending = false;
    m_pendingDx = m_pendingDy = 0.0;
    m_hasLastTs = false;
    emit sampleReady(m_xMm, m_yMm, m_heightMm, 0.0, 0.0, m_squal);
}
```

```cpp
bool SerialPortManager::openSerialPort(const QString& portName, qint32 baudRate) {
	serialPort->setPortName(portName);
	serialPort->setBaudRate(baudRate);
	serialPort->setDataBits(QSerialPort::Data8);
	serialPort->setParity(QSerialPort::NoParity);
	serialPort->setStopBits(QSerialPort::OneStop);
	serialPort->setFlowControl(QSerialPort::NoFlowControl);

	if (serialPort->open(QIODevice::ReadWrite)) {
		qDebug() << "Serial port opened successfully.";
		return true;
	}
	else {
		qDebug() << "Failed to open serial port:" << serialPort->errorString();

	}

		serialPort->close();
		qDebug() << "Serial port closed.";
	}
}

void SerialPortManager::writeData(const QByteArray& data) {
	if (serialPort->isOpen()) {
		serialPort->write(data);
	}
}

QStringList SerialPortManager::getAvailablePorts() const {
	QStringList portList;
	foreach(const QSerialPortInfo & info, QSerialPortInfo::availablePorts()) {
		portList << info.portName();
	}
	return portList;
}

void SerialPortManager::handleReadyRead() {
    m_recvBuf.append(serialPort->readAll()); // 追加到缓冲区

    // 循环查找换行符 \n，分割完整行
    while(true)
    {
        int idx = m_recvBuf.indexOf('\n');
        if(idx < 0) break; // 没有完整一行，退出，等待下次数据

        // 取出一行（包含 \r）
        QByteArray line = m_recvBuf.left(idx+1);
        m_recvBuf = m_recvBuf.mid(idx+1); // 缓冲区剩余数据保留

        line = line.trimmed(); // 去掉 \r \n 回车换行
        if(!line.isEmpty())
        {
            emit dataReceived(line); // 发出**完整一行**
        }
    }
}
```
